// Chạy AI kiểm tra ngữ cảnh và sửa từ bị sai theo từng Lô (8 đoạn/lần) // Giúp tốc độ phản hồi cực nhanh (~2 giây/lô) và không bao giờ bị lỗi lệch câu async function triggerAutoAIPolishBackground(activeScrapeId) { const enabled = document.getElementById("autoAIPolishToggle").checked; const keys = getApiKeysList(); if (!enabled || keys.length === 0 || !chunks.length) return; const label = document.getElementById("aiStatusBtnLabel"); const BATCH_SIZE = 8; const totalBatches = Math.ceil(chunks.length / BATCH_SIZE); let lastUsedKeyNum = 1; for (let b = 0; b < totalBatches; b++) { if (activeScrapeId !== scrapeSessionId) return; const startIdx = b * BATCH_SIZE; const endIdx = Math.min(startIdx + BATCH_SIZE, chunks.length); const batchChunks = chunks.slice(startIdx, endIdx); label.innerText = `✨ AI đang sửa (${b + 1}/${totalBatches})...`; try { const res = await fetch("/api/ai-polish", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chunks: batchChunks, api_keys: keys, start_key_index: currentKeyRotationIndex, }), }); const data = await res.json(); if (activeScrapeId !== scrapeSessionId) return; if (res.ok && Array.isArray(data.chunks) && data.chunks.length === batchChunks.length) { currentKeyRotationIndex = data.next_key_index || 0; lastUsedKeyNum = (data.used_key_index ?? 0) + 1; renderApiKeyList(); for (let offset = 0; offset < data.chunks.length; offset++) { const globalIdx = startIdx + offset; chunks[globalIdx] = data.chunks[offset]; const domChunk = document.getElementById(`chunk_${globalIdx}`); if (domChunk) domChunk.innerText = data.chunks[offset]; // Nếu đoạn này chưa được đọc tới, xóa cache âm thanh cũ để nạp lại giọng đọc mới đã sửa if (globalIdx > currentIndex || !isPlaying) { const oldKey = getCacheKey(globalIdx); if (preloadCache.has(oldKey)) { preloadCache.get(oldKey).then((u) => URL.revokeObjectURL(u)).catch(() => {}); preloadCache.delete(oldKey); } } } // Nạp lại âm thanh cho 5 đoạn sắp đọc if (isPlaying) { for (let ahead = 1; ahead <= PRELOAD_AHEAD; ahead++) { schedulePreload(currentIndex + ahead); } } } } catch (e) { console.warn("Lỗi lô AI:", e); } } if (activeScrapeId === scrapeSessionId) { label.innerText = `✨ AI Đã sửa 100% (Key #${lastUsedKeyNum})`; } }
